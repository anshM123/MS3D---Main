# 2. CENTERED JVP
#
# Since the spectral NS RHS is quadratic, the centered JVP is algebraically
# exact apart from floating-point roundoff:
#
#     DF(U)V =
#       ||V|| [F(U+eps Vhat)-F(U-eps Vhat)] / (2 eps)
#
# ==================================================================================================

@torch.no_grad()
def pair_jvp_L(
    U1,
    U2,
    V1,
    V2,
    env,
):

    vn = pair_norm_D(
        V1,
        V2,
    )

    vn_float = float(
        vn.detach().cpu()
    )

    if (
        not np.isfinite(
            vn_float
        )
        or
        vn_float
        <
        1e-13
    ):

        return (
            torch.zeros_like(U1),
            torch.zeros_like(U2),
        )

    v1 = V1 / vn
    v2 = V2 / vn

    un = pair_norm_D(
        U1,
        U2,
    )

    eps = max(
        1e-5,
        1e-3
        *
        float(
            un.detach().cpu()
        ),
    )

    Fp1, Fp2 = _pair_rhs_D(
        U1 + eps * v1,
        U2 + eps * v2,
        env,
    )

    Fm1, Fm2 = _pair_rhs_D(
        U1 - eps * v1,
        U2 - eps * v2,
        env,
    )

    Jv1 = (
        Fp1 - Fm1
    ) / (
        2.0 * eps
    )

    Jv2 = (
        Fp2 - Fm2
    ) / (
        2.0 * eps
    )

    return (
        vn * Jv1,
        vn * Jv2,
    )

@torch.no_grad()
def rank2_residual_L(
    A1,
    A2,
    model2,
):

    tids = model2[
        "proj"
    ][
        "target_ids"
    ]

    # Remove resolved target P.
    R1, R2 = zero_target_modes_D(
        A1,
        A2,
        tids,
    )

    # Remove q0.
    a0 = pair_inner_D(
        model2["q0_U1"],
        model2["q0_U2"],
        R1,
        R2,
    )

    R1 = (
        R1
        -
        a0
        *
        model2["q0_U1"]
    )

    R2 = (
        R2
        -
        a0
        *
        model2["q0_U2"]
    )

    # Remove q1.
    a1 = pair_inner_D(
        model2["q1_U1"],
        model2["q1_U2"],
        R1,
        R2,
    )

    R1 = (
        R1
        -
        a1
        *
        model2["q1_U1"]
    )

    R2 = (
        R2
        -
        a1
        *
        model2["q1_U2"]
    )

    # Remove numerical target leakage one more time.
    R1, R2 = zero_target_modes_D(
        R1,
        R2,
        tids,
    )

    return R1, R2

@torch.no_grad()
def initialize_model3_L(
    U10,
    U20,
    projector_name,
    env,
):

    # ------------------------------------------------------------------
    # Start with IDENTICAL q0/q1 M3D-2 construction.
    # ------------------------------------------------------------------

    model = initialize_model_D(
        U10,
        U20,
        projector_name,
        env,
    )

    tids = model[
        "proj"
    ][
        "target_ids"
    ]

    dids = model[
        "proj"
    ][
        "dep_ids"
    ]

    # ------------------------------------------------------------------
    # f = F(U0)
    # ------------------------------------------------------------------

    f1, f2 = _pair_rhs_D(
        U10,
        U20,
        env,
    )

    # ------------------------------------------------------------------
    # A f = DF(U0) F(U0)
    # ------------------------------------------------------------------

    Af1, Af2 = pair_jvp_L(
        U10,
        U20,
        f1,
        f2,
        env,
    )

    # ------------------------------------------------------------------
    # r2 = R2 A f
    # ------------------------------------------------------------------

    r21, r22 = rank2_residual_L(
        Af1,
        Af2,
        model,
    )

    r2_norm = pair_norm_D(
        r21,
        r22,
    )

    Af_norm = pair_norm_D(
        Af1,
        Af2,
    )

    r2_float = float(
        r2_norm.detach().cpu()
    )

    Af_float = float(
        Af_norm.detach().cpu()
    )

    # ------------------------------------------------------------------
    # q2* = r2 / ||r2||
    # ------------------------------------------------------------------

    q2_active = bool(
        np.isfinite(
            r2_float
        )
        and
        r2_float
        >
        max(
            1e-11,
            1e-8
            *
            max(
                Af_float,
                1e-30,
            ),
        )
    )

    if q2_active:

        q21 = (
            r21
            /
            r2_norm
        )

        q22 = (
            r22
            /
            r2_norm
        )

        # --------------------------------------------------------------
        # One explicit cleanup orthogonalization against q0, q1 and P.
        # --------------------------------------------------------------

        q21, q22 = zero_target_modes_D(
            q21,
            q22,
            tids,
        )

        b0 = pair_inner_D(
            model["q0_U1"],
            model["q0_U2"],
            q21,
            q22,
        )

        q21 = (
            q21
            -
            b0
            *
            model["q0_U1"]
        )

        q22 = (
            q22
            -
            b0
            *
            model["q0_U2"]
        )

        b1 = pair_inner_D(
            model["q1_U1"],
            model["q1_U2"],
            q21,
            q22,
        )

        q21 = (
            q21
            -
            b1
            *
            model["q1_U1"]
        )

        q22 = (
            q22
            -
            b1
            *
            model["q1_U2"]
        )

        q21, q22 = zero_target_modes_D(
            q21,
            q22,
            tids,
        )

        nq2 = pair_norm_D(
            q21,
            q22,
        )

        q21 = q21 / nq2
        q22 = q22 / nq2

    else:

        q21 = torch.zeros_like(
            model["q0_U1"]
        )

        q22 = torch.zeros_like(
            model["q0_U2"]
        )

    # ------------------------------------------------------------------
    # Verify R3 A f.
    #
    # r3 = r2 - q2 q2* r2
    # ------------------------------------------------------------------

    if q2_active:

        gamma = pair_inner_D(
            q21,
            q22,
            r21,
            r22,
        )

        r31 = (
            r21
            -
            gamma
            *
            q21
        )

        r32 = (
            r22
            -
            gamma
            *
            q22
        )

    else:

        r31 = r21.clone()
        r32 = r22.clone()

    r3_norm = pair_norm_D(
        r31,
        r32,
    )

    r3_ratio = float(
        (
            r3_norm
            /
            (
                r2_norm
                +
                1e-30
            )
        ).detach().cpu()
    )

    # ------------------------------------------------------------------
    # Verify the actual target cubic coefficient:
    #
    #     C3^(2) = P A r2
    #     C3^(3) = P A r3
    #
    # ------------------------------------------------------------------

    Ar21, Ar22 = pair_jvp_L(
        U10,
        U20,
        r21,
        r22,
        env,
    )

    Ar31, Ar32 = pair_jvp_L(
        U10,
        U20,
        r31,
        r32,
        env,
    )

    C32 = pack_modes_D(
        Ar21,
        Ar22,
        tids,
    )

    C33 = pack_modes_D(
        Ar31,
        Ar32,
        tids,
    )

    C32_norm = float(
        torch.linalg.vector_norm(
            C32
        ).detach().cpu()
    )

    C33_norm = float(
        torch.linalg.vector_norm(
            C33
        ).detach().cpu()
    )

    if (
        C32_norm
        >
        1e-12
    ):

        cubic_ratio = (
            C33_norm
            /
            C32_norm
        )

    else:

        cubic_ratio = np.nan

    # ------------------------------------------------------------------
    # Full q0/q1/q2 Gram diagnostic.
    # ------------------------------------------------------------------

    qs = [
        (
            model["q0_U1"],
            model["q0_U2"],
        ),
        (
            model["q1_U1"],
            model["q1_U2"],
        ),
    ]

    if q2_active:
        qs.append(
            (
                q21,
                q22,
            )
        )

    gram_full = torch.zeros(
        (
            len(qs),
            len(qs),
        ),
        device=DEVICE_D,
        dtype=REAL_DTYPE_D,
    )

    for i in range(
        len(qs)
    ):

        for j in range(
            len(qs)
        ):

            gram_full[
                i,
                j
            ] = pair_inner_D(
                qs[i][0],
                qs[i][1],
                qs[j][0],
                qs[j][1],
            )

    gram_error3 = float(
        torch.max(
            torch.abs(
                gram_full
                -
                torch.eye(
                    len(qs),
                    device=DEVICE_D,
                    dtype=REAL_DTYPE_D,
                )
            )
        ).detach().cpu()
    )

    # ------------------------------------------------------------------
    # Extend the stored model.
    # ------------------------------------------------------------------

    model[
        "name"
    ] = (
        f"M3D-3star-{projector_name}"
    )

    model[
        "q2_U1"
    ] = q21

    model[
        "q2_U2"
    ] = q22

    model[
        "q2_active"
    ] = q2_active

    model[
        "effective_hidden_rank"
    ] = (
        3
        if q2_active
        else
        2
    )

    # reset hidden coordinates to the NEW local anchor
    model[
        "z"
    ] = torch.zeros(
        3,
        device=DEVICE_D,
        dtype=REAL_DTYPE_D,
    )

    q0_dep = pack_modes_D(
        model["q0_U1"],
        model["q0_U2"],
        dids,
    )

    q1_dep = pack_modes_D(
        model["q1_U1"],
        model["q1_U2"],
        dids,
    )

    q2_dep = pack_modes_D(
        q21,
        q22,
        dids,
    )

    model[
        "Qr3"
    ] = torch.stack(
        (
            q0_dep,
            q1_dep,
            q2_dep,
        ),
        dim=1,
    )

    model[
        "r2_norm"
    ] = r2_float

    model[
        "r3_ratio"
    ] = r3_ratio

    model[
        "c3_rank2_norm"
    ] = C32_norm

    model[
        "c3_rank3_norm"
    ] = C33_norm

    model[
        "cubic_ratio"
    ] = cubic_ratio

    model[
        "gram_error3"
    ] = gram_error3

    model[
        "cubic_cancel_ok"
    ] = bool(
        (
            not q2_active
        )
        or
        (
            r3_ratio
            <
            R3_RATIO_GATE_L
            and
            (
                not np.isfinite(
                    cubic_ratio
                )
                or
                cubic_ratio
                <
                CUBIC_RATIO_GATE_L
            )
        )
    )

    return model

@torch.no_grad()
def reconstruct_m3d3_L(
    x,
    z,
    U10,
    U20,
    model,
    env,
):

    tids = model[
        "proj"
    ][
        "target_ids"
    ]

    x_phys = (
        x
        *
        model["x_scale"]
    )

    U1, U2 = unpack_modes_D(
        x_phys,
        U10,
        U20,
        tids,
    )

    hs = model[
        "hidden_scale"
    ]

    U1 = (
        U1
        +
        hs
        *
        (
            z[0]
            *
            model["q0_U1"]
            +
            z[1]
            *
            model["q1_U1"]
            +
            z[2]
            *
            model["q2_U1"]
        )
    )

    U2 = (
        U2
        +
        hs
        *
        (
            z[0]
            *
            model["q0_U2"]
            +
            z[1]
            *
            model["q1_U2"]
            +
            z[2]
            *
            model["q2_U2"]
        )
    )

    U1 = dealias_D(
        project_divfree_D(
            U1,
            env,
        ),
        env,
    )

    U2 = dealias_D(
        project_divfree_D(
            U2,
            env,
        ),
        env,
    )

    return U1, U2

@torch.no_grad()
def m3d3_rhs_L(
    x,
    z,
    U10,
    U20,
    model,
    env,
):

    U1, U2 = reconstruct_m3d3_L(
        x,
        z,
        U10,
        U20,
        model,
        env,
    )

    F1, F2 = _pair_rhs_D(
        U1,
        U2,
        env,
    )

    tids = model[
        "proj"
    ][
        "target_ids"
    ]

    dx_phys_dt = pack_modes_D(
        F1,
        F2,
        tids,
    )

    dx_dtau = (
        TAU0_D
        *
        dx_phys_dt
        /
        model["x_scale"]
    )

    hs = model[
        "hidden_scale"
    ]

    dz0 = (
        pair_inner_D(
            model["q0_U1"],
            model["q0_U2"],
            F1,
            F2,
        )
        /
        hs
    )

    dz1 = (
        pair_inner_D(
            model["q1_U1"],
            model["q1_U2"],
            F1,
            F2,
        )
        /
        hs
    )

    dz2 = (
        pair_inner_D(
            model["q2_U1"],
            model["q2_U2"],
            F1,
            F2,
        )
        /
        hs
    )

    dz_dtau = (
        TAU0_D
        *
        torch.stack(
            (
                dz0,
                dz1,
                dz2,
            )
        )
    ).to(
        REAL_DTYPE_D
    )

    return (
        dx_dtau.to(
            REAL_DTYPE_D
        ),
        dz_dtau,
    )

@torch.no_grad()
def m3d3_rk4_L(
    model,
    d_tau,
    env,
):

    x0 = model["x"]
    z0 = model["z"]

    U10 = model[
        "anchor_U1"
    ]

    U20 = model[
        "anchor_U2"
    ]

    k1x, k1z = m3d3_rhs_L(
        x0,
        z0,
        U10,
        U20,
        model,
        env,
    )

    k2x, k2z = m3d3_rhs_L(
        x0
        +
        0.5
        *
        d_tau
        *
        k1x,

        z0
        +
        0.5
        *
        d_tau
        *
        k1z,

        U10,
        U20,
        model,
        env,
    )

    k3x, k3z = m3d3_rhs_L(
        x0
        +
        0.5
        *
        d_tau
        *
        k2x,

        z0
        +
        0.5
        *
        d_tau
        *
        k2z,

        U10,
        U20,
        model,
        env,
    )

    k4x, k4z = m3d3_rhs_L(
        x0
        +
        d_tau
        *
        k3x,

        z0
        +
        d_tau
        *
        k3z,

        U10,
        U20,
        model,
        env,
    )

    model["x"] = (
        x0
        +
        (
            d_tau
            /
            6.0
        )
        *
        (
            k1x
            +
            2.0 * k2x
            +
            2.0 * k3x
            +
            k4x
        )
    )

    model["z"] = (
        z0
        +
        (
            d_tau
            /
            6.0
        )
        *
        (
            k1z
            +
            2.0 * k2z
            +
            2.0 * k3z
            +
            k4z
        )
    )

    return (
        model["x"],
        model["z"],
    )

def relative_error_L(
    a,
    b,
):

    return float(
        (
            torch.linalg.vector_norm(
                a - b
            )
            /
            (
                torch.linalg.vector_norm(
                    b
                )
                +
                1e-30
            )
        ).detach().cpu()
    )


def finite_model3_L(
    model,
):

    return bool(
        torch.isfinite(
            model["x"]
        ).all().item()
        and
        torch.isfinite(
            model["z"]
        ).all().item()
    )


@torch.no_grad()
def model_target3_L(
    model,
):

    return (
        model["x"]
        *
        model["x_scale"]
    )


@torch.no_grad()
def truth_target_L(
    U1,
    U2,
    model,
):

    return pack_modes_D(
        U1,
        U2,
        model[
            "proj"
        ][
            "target_ids"
        ],
    )


@torch.no_grad()
def truth_target_rhs_L(
    U1,
    U2,
    model,
    nu,
    env,
):

    F1 = velocity_rhs_D(
        U1,
        nu,
        env,
    )

    F2 = velocity_rhs_D(
        U2,
        nu,
        env,
    )

    return pack_modes_D(
        F1,
        F2,
        model[
            "proj"
        ][
            "target_ids"
        ],
    )


@torch.no_grad()
def model_target_rhs3_L(
    model,
    env,
):

    dx_tau, _ = m3d3_rhs_L(
        model["x"],
        model["z"],
        model["anchor_U1"],
        model["anchor_U2"],
        model,
        env,
    )

    return (
        dx_tau
        *
        model["x_scale"]
        /
        float(
            TAU0_D
        )
    )


@torch.no_grad()
def score3_L(
    model,
    U1_truth,
    U2_truth,
    nu,
    env,
):

    tx = truth_target_L(
        U1_truth,
        U2_truth,
        model,
    )

    px = model_target3_L(
        model
    )

    trhs = truth_target_rhs_L(
        U1_truth,
        U2_truth,
        model,
        nu,
        env,
    )

    prhs = model_target_rhs3_L(
        model,
        env,
    )

    return (
        relative_error_L(
            px,
            tx,
        ),
        relative_error_L(
            prhs,
            trhs,
        ),
    )

@torch.no_grad()
def run_case3_L(
    topology,
    nu_mult,
    case_number,
):

    global NU1_D
    global NU2_D
    global TAU0_D

    nu = (
        float(
            BASE_NU_K
        )
        *
        float(
            nu_mult
        )
    )

    NU1_D = nu
    NU2_D = nu
    TAU0_D = COMMON_TAU0_L

    U10, U20 = topology_pair_K(
        topology,
        ENV_L,
    )

    # CFD truth is scoring-only.
    U1_truth = U10.clone()
    U2_truth = U20.clone()

    torch.cuda.synchronize()

    build_start = time.perf_counter()

    model = initialize_model3_L(
        U10,
        U20,
        PROJECTOR_K,
        ENV_L,
    )

    torch.cuda.synchronize()

    initial_build_s = (
        time.perf_counter()
        -
        build_start
    )

    if not model[
        "cubic_cancel_ok"
    ]:

        raise RuntimeError(
            "Initial cubic cancellation unexpectedly failed."
        )

    next_anchor_tau = (
        ANCHOR_INTERVAL_L
    )

    next_report_i = 0

    reanchors = 0
    rebuild_total_s = 0.0

    worst_jump = 0.0
    worst_r3_ratio = float(
        model[
            "r3_ratio"
        ]
    )

    finite_cubic_ratios = []

    if np.isfinite(
        model[
            "cubic_ratio"
        ]
    ):
        finite_cubic_ratios.append(
            float(
                model[
                    "cubic_ratio"
                ]
            )
        )

    event_rows = []
    checkpoint_rows = []

    torch.cuda.synchronize()

    trajectory_start = time.perf_counter()

    final_tau = 0.0

    for step in range(
        1,
        MAX_STEPS_L + 1,
    ):

        # ------------------------------------------------------------------
        # Full NS truth — never used by reanchoring.
        # ------------------------------------------------------------------

        U1_truth = full_rk4_D(
            U1_truth,
            DT_PHYS_L,
            nu,
            ENV_L,
        )

        U2_truth = full_rk4_D(
            U2_truth,
            DT_PHYS_L,
            nu,
            ENV_L,
        )

        # ------------------------------------------------------------------
        # M3D-3*
        # ------------------------------------------------------------------

        m3d3_rk4_L(
            model,
            DTAU_L,
            ENV_L,
        )

        current_tau = (
            step
            *
            DTAU_L
        )

        final_tau = current_tau

        if not finite_model3_L(
            model
        ):

            print(
                "MODEL BECAME NON-FINITE."
            )

            break

        # ------------------------------------------------------------------
        # Teacher-free periodic reanchor.
        # ------------------------------------------------------------------

        if (
            current_tau
            >=
            next_anchor_tau
            -
            0.5
            *
            DTAU_L
            and
            next_anchor_tau
            <
            FINAL_TAU_L
            -
            1e-12
        ):

            target_before = (
                model_target3_L(
                    model
                ).clone()
            )

            torch.cuda.synchronize()

            rb_start = time.perf_counter()

            # ==============================================================
            # NO TRUTH STATE HERE.
            # ==============================================================

            model = reanchor_model3_L(
                model,
                ENV_L,
            )

            # ==============================================================

            torch.cuda.synchronize()

            rb_s = (
                time.perf_counter()
                -
                rb_start
            )

            target_after = model_target3_L(
                model
            )

            jump = relative_error_L(
                target_after,
                target_before,
            )

            if not model[
                "cubic_cancel_ok"
            ]:

                raise RuntimeError(
                    "\nCUBIC CANCELLATION FAILED DURING REANCHOR.\n"
                    f"case={case_number}\n"
                    f"topology={topology}\n"
                    f"nu_mult={nu_mult}\n"
                    f"tau={current_tau}\n"
                    f"R3/R2={model['r3_ratio']}\n"
                    f"C3 ratio={model['cubic_ratio']}"
                )

            reanchors += 1

            rebuild_total_s += (
                rb_s
            )

            worst_jump = max(
                worst_jump,
                jump,
            )

            worst_r3_ratio = max(
                worst_r3_ratio,
                float(
                    model[
                        "r3_ratio"
                    ]
                ),
            )

            if np.isfinite(
                model[
                    "cubic_ratio"
                ]
            ):

                finite_cubic_ratios.append(
                    float(
                        model[
                            "cubic_ratio"
                        ]
                    )
                )

            event_rows.append(
                {
                    "case":
                        case_number,

                    "topology":
                        topology,

                    "nu_mult":
                        nu_mult,

                    "reanchor":
                        reanchors,

                    "tau":
                        current_tau,

                    "scheduled_tau":
                        next_anchor_tau,

                    "build_seconds":
                        rb_s,

                    "target_jump":
                        jump,

                    "r2_norm":
                        model[
                            "r2_norm"
                        ],

                    "r3_ratio":
                        model[
                            "r3_ratio"
                        ],

                    "c3_rank2_norm":
                        model[
                            "c3_rank2_norm"
                        ],

                    "c3_rank3_norm":
                        model[
                            "c3_rank3_norm"
                        ],

                    "cubic_ratio":
                        model[
                            "cubic_ratio"
                        ],

                    "gram_error3":
                        model[
                            "gram_error3"
                        ],
                }
            )

            next_anchor_tau += (
                ANCHOR_INTERVAL_L
            )

        # ------------------------------------------------------------------
        # Reports.
        # ------------------------------------------------------------------

        while (
            next_report_i
            <
            len(
                REPORT_TAUS_L
            )
            and
            current_tau
            >=
            REPORT_TAUS_L[
                next_report_i
            ]
            -
            0.5
            *
            DTAU_L
        ):

            state_err, rhs_err = score3_L(
                model,
                U1_truth,
                U2_truth,
                nu,
                ENV_L,
            )

            checkpoint_rows.append(
                {
                    "case":
                        case_number,

                    "topology":
                        topology,

                    "nu_mult":
                        nu_mult,

                    "tau":
                        current_tau,

                    "state_error":
                        state_err,

                    "rhs_error":
                        rhs_err,

                    "reanchors":
                        reanchors,
                }
            )

            print(
                f"    tau={current_tau:6.3f}"
                f" | M3D-3* state={100*state_err:9.4f}%"
                f" | RHS={100*rhs_err:9.4f}%"
                f" | anchors={reanchors:2d}"
            )

            next_report_i += 1

    torch.cuda.synchronize()

    trajectory_s = (
        time.perf_counter()
        -
        trajectory_start
    )

    state_final, rhs_final = score3_L(
        model,
        U1_truth,
        U2_truth,
        nu,
        ENV_L,
    )

    strict_pass = bool(
        finite_model3_L(
            model
        )
        and
        final_tau
        >=
        FINAL_TAU_L
        -
        DTAU_L
        and
        state_final
        <
        0.01
        and
        rhs_final
        <
        0.10
    )

    return (
        {
            "case":
                case_number,

            "topology":
                topology,

            "nu_mult":
                float(
                    nu_mult
                ),

            "final_tau":
                final_tau,

            "state_error_m3d3":
                state_final,

            "rhs_error_m3d3":
                rhs_final,

            "initial_build_s":
                initial_build_s,

            "reanchors":
                reanchors,

            "rebuild_total_s":
                rebuild_total_s,

            "mean_rebuild_s":
                (
                    rebuild_total_s
                    /
                    max(
                        reanchors,
                        1,
                    )
                ),

            "worst_target_jump":
                worst_jump,

            "worst_r3_ratio":
                worst_r3_ratio,

            "worst_cubic_ratio":
                (
                    max(
                        finite_cubic_ratios
                    )
                    if
                    len(
                        finite_cubic_ratios
                    )
                    else
                    np.nan
                ),

            "trajectory_runtime_s":
                trajectory_s,

            "strict_pass":
                strict_pass,
        },

        checkpoint_rows,

        event_rows,
    )
